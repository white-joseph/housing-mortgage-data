# Load dataset

df <- read_dta(here("data", "clean", "uad_clean.dta"))

df %>%
  group_by(program, year) %>%
  summarise(non_null_appraisals = sum(!is.na(appraised_value)), .groups = "drop") %>%
  ggplot(aes(x = year, y = non_null_appraisals, fill = program)) +
    geom_col() +
    scale_x_continuous(breaks = 2013:2022) +
    facet_wrap(~ program, scales = "free_x") +
    labs(x = "Year", y = "Count of Non-Null Appraisals", title = "Appraisals by Year and Program") +
    theme_minimal() +
    theme(axis.text.x = element_text(angle = 45, hjust = 1))

Appraisal to contract ratio

df %>%

Property characterstics

df %>%
  select(quality, condition, bedrooms, bathrooms, lot_size, gross_living_area) %>%
  mutate(across(everything(), as_factor)) %>%
  pivot_longer(cols = everything(), names_to = "Variable", values_to = "Value") %>%
  ggplot(aes(x = Value)) +
    geom_bar(fill = "steelblue") +
    facet_wrap(~ Variable, scales = "free") +
    scale_y_continuous(labels = scales::comma) +
    theme_minimal() +
    labs(title = "Distribution of Property Characteristics")
